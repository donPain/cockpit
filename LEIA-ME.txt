                    INSTRUÇÕES DE INSTALAÇÃO
              COMMAND CENTER - LAUNCHER DE PERFIS CHROME
═══════════════════════════════════════════════════════════════

📁 PASSO 1: ORGANIZAR ARQUIVOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Crie uma pasta (ex: C:\CommandCenter\) e coloque TODOS estes arquivos:

✓ cockpit.html
✓ perfil-stg-br.bat
✓ perfil-sandro-master.bat
✓ perfil-propoint.bat
✓ perfil-ludus.bat
✓ perfil-strike.bat
✓ perfil-stg-us.bat


🔍 PASSO 2: DESCOBRIR OS NOMES DOS PERFIS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Abra o Chrome
2. Digite na barra de endereço: chrome://version/
3. Procure a linha "Profile Path"
4. Você verá algo assim:

   C:\Users\SeuNome\AppData\Local\Google\Chrome\User Data\Profile 2
                                                           ^^^^^^^^
                                                     ESTE É O NOME!

Os nomes podem ser:
   • Default
   • Profile 1
   • Profile 2
   • Profile 3
   • etc.


✏️ PASSO 3: EDITAR OS ARQUIVOS .BAT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Clique com BOTÃO DIREITO em cada arquivo .bat → Editar

ANTES (exemplo):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
@echo off
start "" "C:\Program Files\Google\Chrome\Application\chrome.exe" --profile-directory="Profile 1"
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

DEPOIS (com o nome correto do seu perfil):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
@echo off
start "" "C:\Program Files\Google\Chrome\Application\chrome.exe" --profile-directory="Default"
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                                                                  ^^^^^^^^
                                                        MUDE PARA O NOME CERTO!


📋 EXEMPLO DE CONFIGURAÇÃO COMPLETA:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

perfil-stg-br.bat         → "Default"
perfil-sandro-master.bat  → "Profile 1" 
perfil-propoint.bat       → "Profile 2"
perfil-ludus.bat          → "Profile 3"
perfil-strike.bat         → "Profile 4"
perfil-stg-us.bat         → "Profile 5"


🚀 PASSO 4: USAR O COMMAND CENTER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Dê DUPLO CLIQUE no cockpit.html
2. Clique nos botões dos perfis
3. O Chrome abrirá com o perfil correspondente!


🔧 RESOLUÇÃO DE PROBLEMAS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ PROBLEMA: Nada acontece ao clicar
✓ SOLUÇÃO: Verifique se o Chrome está instalado em:
   C:\Program Files\Google\Chrome\Application\chrome.exe
   
   Se estiver em outro lugar, edite TODOS os .bat com o caminho correto.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ PROBLEMA: Abre perfil errado
✓ SOLUÇÃO: Confira o nome do perfil em chrome://version/ e corrija o .bat

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ PROBLEMA: Aparece janela do Windows Defender
✓ SOLUÇÃO: É normal. Clique em "Mais informações" → "Executar assim mesmo"
   Ou adicione a pasta à lista de exceções do antivírus.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Clique com botão direito em cockpit.html
2. Enviar para → Área de trabalho (criar atalho)
3. Agora você tem acesso rápido!
